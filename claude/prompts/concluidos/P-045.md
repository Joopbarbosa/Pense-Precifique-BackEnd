### Endpoints:

```
GET    /clientes              → listar (paginado, param opcional: nome=)
GET    /clientes/{id}         → buscar por id
POST   /clientes              → cadastrar
PUT    /clientes/{id}         → editar
DELETE /clientes/{id}         → inativar (soft delete)
```

### Implementação:

```java
@RestController
@RequestMapping("/clientes")
@RequiredArgsConstructor
public class ClienteController {

    private final ClienteService clienteService;

    @GetMapping
    public ResponseEntity<Page<ClienteResponse>> listar(
            @AuthenticationPrincipal UserDetails userDetails,
            @RequestParam(required = false) String nome,
            @PageableDefault(size = 20, sort = "nome") Pageable pageable) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(clienteService.listar(usuarioId, nome, pageable));
    }

    @GetMapping("/{id}")
    public ResponseEntity<ClienteResponse> buscar(
            @PathVariable UUID id,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(clienteService.buscarPorId(id, usuarioId));
    }

    @PostMapping
    public ResponseEntity<ClienteResponse> cadastrar(
            @Valid @RequestBody ClienteRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.status(201).body(clienteService.cadastrar(request, usuarioId));
    }

    @PutMapping("/{id}")
    public ResponseEntity<ClienteResponse> editar(
            @PathVariable UUID id,
            @Valid @RequestBody ClienteRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(clienteService.editar(id, request, usuarioId));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> inativar(
            @PathVariable UUID id,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        clienteService.inativar(id, usuarioId);
        return ResponseEntity.noContent().build();
    }
}
```

Reutilize o mesmo padrão de extração de `usuarioId` dos controllers anteriores.
