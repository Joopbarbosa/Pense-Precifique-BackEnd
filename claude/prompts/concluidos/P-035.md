### Endpoints:

```
GET    /produtos              → listar (paginado, param opcional: tipo=PRODUTO|PRODUTO_BASE|CUSTOMIZACAO)
GET    /produtos/{id}         → buscar por id (retorna ProdutoDetalheResponse com ficha técnica)
POST   /produtos              → cadastrar
PUT    /produtos/{id}         → editar
DELETE /produtos/{id}         → inativar (soft delete)
```

### Implementação:

```java
@RestController
@RequestMapping("/produtos")
@RequiredArgsConstructor
public class ProdutoController {

    private final ProdutoService produtoService;

    @GetMapping
    public ResponseEntity<Page<ProdutoResponse>> listar(
            @AuthenticationPrincipal UserDetails userDetails,
            @RequestParam(required = false) TipoProduto tipo,
            @PageableDefault(size = 20, sort = "nome") Pageable pageable) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(produtoService.listar(usuarioId, tipo, pageable));
    }

    @GetMapping("/{id}")
    public ResponseEntity<ProdutoDetalheResponse> buscar(
            @PathVariable UUID id,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(produtoService.buscarPorId(id, usuarioId));
    }

    @PostMapping
    public ResponseEntity<ProdutoDetalheResponse> cadastrar(
            @Valid @RequestBody ProdutoRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.status(201).body(produtoService.cadastrar(request, usuarioId));
    }

    @PutMapping("/{id}")
    public ResponseEntity<ProdutoDetalheResponse> editar(
            @PathVariable UUID id,
            @Valid @RequestBody ProdutoRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(produtoService.editar(id, request, usuarioId));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> inativar(
            @PathVariable UUID id,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        produtoService.inativar(id, usuarioId);
        return ResponseEntity.noContent().build();
    }
}
```

Reutilize o mesmo padrão de extração de `usuarioId` dos controllers anteriores.
