### Endpoints:

```
GET    /insumos              → listar (paginado, param: page=0, size=20)
GET    /insumos/{id}         → buscar por id
POST   /insumos              → cadastrar
PUT    /insumos/{id}         → editar
DELETE /insumos/{id}         → inativar (soft delete)
```

### Implementação:

```java
@RestController
@RequestMapping("/insumos")
@RequiredArgsConstructor
public class InsumoController {

    private final InsumoService insumoService;

    @GetMapping
    public ResponseEntity<Page<InsumoResponse>> listar(
            @AuthenticationPrincipal UserDetails userDetails,
            @PageableDefault(size = 20, sort = "nome") Pageable pageable) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(insumoService.listar(usuarioId, pageable));
    }

    @GetMapping("/{id}")
    public ResponseEntity<InsumoResponse> buscar(
            @PathVariable UUID id,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(insumoService.buscarPorId(id, usuarioId));
    }

    @PostMapping
    public ResponseEntity<InsumoResponse> cadastrar(
            @Valid @RequestBody InsumoRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.status(201).body(insumoService.cadastrar(request, usuarioId));
    }

    @PutMapping("/{id}")
    public ResponseEntity<InsumoResponse> editar(
            @PathVariable UUID id,
            @Valid @RequestBody InsumoRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        return ResponseEntity.ok(insumoService.editar(id, request, usuarioId));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> inativar(
            @PathVariable UUID id,
            @AuthenticationPrincipal UserDetails userDetails) {
        UUID usuarioId = extrairUsuarioId(userDetails);
        insumoService.inativar(id, usuarioId);
        return ResponseEntity.noContent().build();
    }
}
```

### Extração do usuarioId:
